# ---------- Simulación VHDL ----------
ghdl        >= 4.1     # compilador/simulador VHDL. Ubuntu(WSL) trae 4.1, Arch trae 6.0.
                       # EN ARCH NO ESTÁ EN LOS REPOS OFICIALES: vive en AUR
                       #   (paquete `ghdl`, backend mcode, depende de gcc-ada).
                       #   Instalación: yay -S ghdl   (o paru -S ghdl)
                       #   Sin helper AUR: git clone https://aur.archlinux.org/ghdl.git
                       #                   && cd ghdl && makepkg -si
                       # Backend mcode: SIMULACIÓN únicamente (no sirve para síntesis).
                       # Los fuentes deben ser compatibles con ambos => VHDL-2008 conservador.
gtkwave     >= 3.3     # visor de formas de onda (archivos .vcd).
                       # En WSL necesita GUI: WSLg (Windows 11) o X server / visor en Windows.

# ---------- Compilación LaTeX (Springer llncs) ----------
latexmk     >= 4.87    # orquestador de compilación (usado por scripts/latexbuild).
xelatex     # motor Unicode — scripts/latexbuild invoca `latexmk -xelatex`.
                       # OJO: además del binario hace falta el FORMATO xelatex.fmt.
                       # En Arch el binario lo trae `texlive-bin`, pero el formato
                       # lo aporta `texlive-xetex` (sin él: "I can't find the
                       # format file `xelatex.fmt'!"). En Ubuntu: `texlive-xetex`.
                       # Verificación: kpsewhich --engine=xetex xelatex.fmt
                       # (sin --engine, kpsewhich NO busca .fmt y devuelve vacío
                       #  aunque el formato esté instalado).
llncs.cls   # clase Springer LNCS. Paquete: texlive-publishers (Arch/Ubuntu).
splncs04.bst# estilo de bibliografía Springer. Paquete: texlive-publishers.
# NOTA: en Arch y Ubuntu llncs.cls y splncs04.bst vienen en `texlive-publishers`.
# Verificación: kpsewhich llncs.cls && kpsewhich splncs04.bst
fuentes-lm  # Latin Modern en OTF (lmroman10-regular.otf…). NO es opcional:
            # llncs 2025+ carga `fontspec` y pide esa familia. Si falta, xelatex
            # no aborta con un error legible: usa `nullfont` y el reporte sale
            # con 17 páginas en blanco y miles de "Missing character: There is
            # no X (...) in font nullfont!".
            #   Arch:   sudo pacman -S --needed texlive-fontsrecommended
            #   Ubuntu: sudo apt install texlive-fonts-recommended
            # Verificación: kpsewhich lmroman10-regular.otf
            #   (o: fc-list | grep -i "latin modern")
            # El mismo paquete aporta pzdr.tfm (Zapf Dingbats) que hyperref usa
            # para los recuadros de enlace; sin él solo hay un aviso inofensivo.

# ---------- Utilidades ----------
git        >= 2.30    # control de versiones.
make       >= 4.3     # automatización (disponible en todo el equipo).
